import { writeFileSync } from 'node:fs';
import { describe, expect, it } from 'vitest';
import { ATTACK_DIRECTOR } from '../../../src/sim/balance';
import { Match } from '../../../src/sim/match';
import type { Difficulty } from '../../../src/sim/types';

/** Frequency of suspicious situations in natural (autoPlayer) matches. */
describe('QA sim stats', () => {
  it('counts flee-from-entering, respite-by-exclude and pick timing', () => {
    const N = Number(process.env.QA_N ?? 100);
    const out: string[] = [];
    for (const diff of ['easy', 'hard', 'nightmare'] as Difficulty[]) {
      let matches = 0;
      let fleeEntering = 0;
      let brokenAliveRooms = 0;
      let brokenAlivePlayer = 0;
      let enteredNoSiege = 0;
      let respiteViol = 0;
      let respiteViolSamples: string[] = [];
      let lostPlayerMatches = 0;
      for (let s = 1; s <= N; s++) {
        const m = new Match({ seed: s * 31 + 7, difficulty: diff, flameUnlocked: true, autoPlayer: true, autoPlayerSkill: 0.5 });
        matches++;
        let prevState = m.ghost.state;
        let lastPlayerSiege = -1e9;
        const brokenSeen = new Set<number>();
        let ticks = 0;
        while (m.phase !== 'end' && ticks++ < 20 * 60 * 40) {
          m.step();
          const g = m.ghost;
          for (const e of m.events) {
            if (e.type === 'ghostRetreat' && prevState === 'entering') fleeEntering++;
            if (e.type === 'siegeEnd' && e.meaningful && m.rooms[e.roomId].ownerId === 0) lastPlayerSiege = m.nightTime;
            if (e.type === 'ghostTarget' && m.rooms[e.roomId].ownerId === 0) {
              const since = m.nightTime - lastPlayerSiege;
              const others = m.rooms.filter((r) => r.ownerId !== null && r.ownerId !== 0 && !r.eliminated).length;
              if (since < ATTACK_DIRECTOR[diff].respite && others > 0) {
                respiteViol++;
                if (respiteViolSamples.length < 3) respiteViolSamples.push(`seed ${s * 31 + 7} t=${m.nightTime.toFixed(1)} since=${since.toFixed(1)} others=${others}`);
              }
            }
          }
          // rooms alive with a broken door while the ghost is not entering them
          for (const r of m.rooms) {
            if (r.ownerId === null || r.eliminated || !r.door.broken) continue;
            if (g.state === 'entering' && g.targetRoom === r.id) continue;
            if (!brokenSeen.has(r.id)) {
              brokenSeen.add(r.id);
              brokenAliveRooms++;
              if (r.ownerId === 0) brokenAlivePlayer++;
            }
          }
          if (g.state === 'entering' && prevState === 'moving') enteredNoSiege++;
          prevState = g.state;
        }
        if (m.player.caught) lostPlayerMatches++;
      }
      out.push(
        `${diff}: matches=${matches} fleeFromEntering=${fleeEntering} aliveRoomsWithBrokenDoor=${brokenAliveRooms} (player ${brokenAlivePlayer}) ghostWalkedStraightInNoSiege=${enteredNoSiege} respiteViolations=${respiteViol} playerCaught=${lostPlayerMatches} ${respiteViolSamples.join(' ; ')}`,
      );
    }
    if (process.env.QA_OUT) writeFileSync(process.env.QA_OUT, out.join(String.fromCharCode(10)));
    else console.log(out.join(String.fromCharCode(10)));
    expect(true).toBe(true);
  }, 900_000);

  it('neighbours pick rooms all in the same tick when the player picks manually', () => {
    const m = new Match({ seed: 3, difficulty: 'easy', flameUnlocked: true });
    m.step();
    expect(m.command(0, { type: 'pickRoom', roomId: 1 })).toBeNull();
    m.step();
    const picked = m.chars.filter((c) => !c.isPlayer && c.roomId !== null).length;
    const m2 = new Match({ seed: 3, difficulty: 'easy', flameUnlocked: true });
    while (m2.player.roomId === null) m2.step(); // auto-assigned at pick timeout
    m2.step();
    const picked2 = m2.chars.filter((c) => !c.isPlayer && c.roomId !== null).length;
    const msg = `SIM-pick manual: npcs picked 1 tick after player=${picked}, phase=${m.phase}; auto-assign: npcs picked 1 tick after=${picked2}, phase=${m2.phase}`;
    if (process.env.QA_OUT2) writeFileSync(process.env.QA_OUT2, msg);
    else console.log(msg);
    expect(picked).toBeLessThan(5); // FAILS: all 5 at once
  });
});

describe('QA natural seeds', () => {
  it('natural repro: SIM-1 (hard seed 100) and SIM-2 flee-from-entering seeds', () => {
    const lines: string[] = [];
    for (const [diff, seed] of [['hard', 100], ['nightmare', 1743]] as const) {
      const m = new Match({ seed, difficulty: diff, flameUnlocked: true, autoPlayer: true, autoPlayerSkill: 0.5 });
      let last = -1e9;
      let prevTarget = -1;
      let ticks = 0;
      while (m.phase !== 'end' && ticks++ < 20 * 60 * 40) {
        m.step();
        for (const e of m.events) {
          if (e.type === 'siegeEnd' && e.meaningful && m.rooms[e.roomId].ownerId === 0) last = m.nightTime;
          if (e.type === 'ghostTarget') {
            if (m.rooms[e.roomId].ownerId === 0 && m.nightTime - last < ATTACK_DIRECTOR[diff].respite) {
              const alive = m.rooms.filter((r) => r.ownerId !== null && !r.eliminated).map((r) => r.id);
              lines.push(`${diff} seed ${seed}: t=${m.nightTime.toFixed(2)} player room ${e.roomId} targeted ${(m.nightTime - last).toFixed(1)}s after its real siege (respite ${ATTACK_DIRECTOR[diff].respite}); previous target=${prevTarget}; alive=${JSON.stringify(alive)}`);
            }
            prevTarget = e.roomId;
          }
        }
      }
    }
    for (let s = 1; s <= 100 && lines.length < 8; s++) {
      const seed = s * 31 + 7;
      const m = new Match({ seed, difficulty: 'hard', flameUnlocked: true, autoPlayer: true, autoPlayerSkill: 0.5 });
      let prev = m.ghost.state;
      let fled = -1;
      let ticks = 0;
      while (m.phase !== 'end' && ticks++ < 20 * 60 * 40) {
        m.step();
        if (m.events.some((e) => e.type === 'ghostRetreat') && prev === 'entering') fled = m.ghost.targetRoom;
        prev = m.ghost.state;
      }
      if (fled >= 0) {
        const r = m.rooms[fled];
        lines.push(`hard seed ${seed}: ghost fled while entering room ${fled} (owner ${r.ownerId}); end: broken=${r.door.broken} eliminated=${r.eliminated} result=${m.result}`);
      }
    }
    const txt = lines.join(String.fromCharCode(10));
    if (process.env.QA_OUT) writeFileSync(process.env.QA_OUT, txt);
    else console.log(txt);
    expect(true).toBe(true);
  }, 600_000);
});
