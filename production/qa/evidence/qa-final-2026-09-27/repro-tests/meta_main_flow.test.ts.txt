import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest';

type Any = any; // eslint-disable-line @typescript-eslint/no-explicit-any
const qa: Any = ((globalThis as Any).__qa = (globalThis as Any).__qa ?? {});

  }
  class Game {
    scale = { getParentBounds() {}, refresh() {} };

      const q = (globalThis as Any).__qa;

          if (k === 'boot') q.BootScene = cls;

          starts.push(d);
        },

        isActive: (k: string) => active.has(k),
        starts,
      };
      q.game = this;
    }
  }
  return { default: { Game, Scene, Scale: { RESIZE: 3 }, Loader: { Events: { COMPLETE: 'complete' } } } };
});

    constructor() {

          if (p in t) return t[p];

            t.calls.push([p, a]);
            if (p.startsWith('show') && p !== 'showMenu') t.screen = [p, a];

          };
        },
      });
    }
  },
}));

// ---------- globals ----------
const docListeners: Record<string, Array<() => void>> = {};
function el(): Any {
  return { classList: { add() {}, remove() {} }, style: { setProperty() {}, cssText: '' }, appendChild() {}, remove() {} };
}
function installGlobals() {
  for (const k of Object.keys(docListeners)) delete docListeners[k];
  vi.stubGlobal('document', {
    addEventListener: (t: string, f: () => void) => (docListeners[t] ??= []).push(f),
    getElementById: () => el(),
    createElement: () => el(),
    body: el(),
    hidden: false,
    visibilityState: 'visible',
  });
  vi.stubGlobal('window', {
    addEventListener() {},
    innerWidth: 800,
    innerHeight: 600,
    setTimeout: (...a: Any[]) => (setTimeout as Any)(...a),
    clearTimeout: (...a: Any[]) => (clearTimeout as Any)(...a),
  });
  vi.stubGlobal('location', { search: '' });
  vi.stubGlobal('screen', {});
  vi.stubGlobal('requestAnimationFrame', () => 0);
  vi.stubGlobal('cancelAnimationFrame', () => {});
  const m = new Map<string, string>();
  vi.stubGlobal('localStorage', { getItem: (k: string) => m.get(k) ?? null, setItem: (k: string, v: string) => void m.set(k, v), removeItem: (k: string) => void m.delete(k) });
}

const today = (() => {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
})();

function snapshot(rev: number, progress: Any) {
  return { v: 2, rev, at: rev, progress };
}
function returning(extra: Any = {}) {
  return { matches: 10, tutorial: 'done', meta: { coins: 100, daily: { last: today, step: 1 } }, ...extra };
}

interface SdkOpts {
  local?: Any;
  authorized?: boolean | (() => boolean);
  getData?: () => Promise<Any>;
}
function makeSdk(o: SdkOpts) {
  const safe = new Map<string, string>();
  if (o.local) safe.set('midnight-door-progress', JSON.stringify(o.local));
  let adCbs: Any = null;
  const adCalls: string[] = [];
  const auth = typeof o.authorized === 'function' ? o.authorized : () => !!o.authorized;
  const player = {
    isAuthorized: auth,
    getData: vi.fn(o.getData ?? (async () => ({}))),
    setData: vi.fn(async () => {}),
  };
  const sdk = {
    getPlayer: vi.fn(async () => player),
    getStorage: async () => ({ getItem: (k: string) => safe.get(k) ?? null, setItem: (k: string, v: string) => void safe.set(k, v), removeItem: (k: string) => void safe.delete(k) }),
    auth: { openAuthDialog: vi.fn(async () => {}) },
    environment: { i18n: { lang: 'ru' } },
    features: { LoadingAPI: { ready: vi.fn() }, GameplayAPI: { start() {}, stop() {} } },
    adv: {
      showFullscreenAdv: ({ callbacks }: Any) => (adCalls.push('fs'), (adCbs = callbacks)),
      showRewardedVideo: ({ callbacks }: Any) => (adCalls.push('rw'), (adCbs = callbacks)),
    },
    on() {},
  };
  return { sdk, player, safe, adCalls, ad: () => adCbs };
}

const flush = () => vi.advanceTimersByTimeAsync(0);
async function bootMain(sdk: Any) {
  qa.sdk = sdk;
  vi.resetModules();
  await import('../../../src/main');
}
const spritesDone = () => new qa.BootScene().create();
const lastCall = (name: string) => [...qa.hud.calls].reverse().find((c: Any) => c[0] === name)?.[1];
const starts = (): Any[] => qa.game.scene.starts;

beforeEach(() => {
  vi.useFakeTimers();
  installGlobals();
});
afterEach(() => {
  vi.useRealTimers();
  vi.unstubAllGlobals();
});

describe('main.ts flow races', () => {
  it('double tap on a difficulty while the pre-match interstitial is pending -> two matches started, final one is the FIRST tap', async () => {
    const f = makeSdk({ local: snapshot(10, returning()) });
    await bootMain(f.sdk);
    await flush();
    spritesDone();
    await flush();
    const onPlay = lastCall('showStart')![1];
    onPlay('easy'); // ad requested, menu still on screen
    onPlay('nightmare'); // second tap reaches the menu (screen did not change -> no echo guard)
    await flush();
    expect(f.adCalls).toEqual(['fs']);
    expect(starts().length).toBe(1);
    expect(starts()[0].match.opts.difficulty).toBe('nightmare'); // started UNDER the ad
    f.ad().onClose(true);
    await flush();
    expect(starts().length).toBe(2);
    expect(starts()[1].match.opts.difficulty).toBe('easy'); // player ends up in the first-tapped difficulty
  });

  it('result screen: "Ещё раз" then "Меню" while the interstitial is pending -> menu shown, then a match starts anyway', async () => {
    const f = makeSdk({ local: snapshot(10, returning()) });
    await bootMain(f.sdk);
    await flush();
    spritesDone();
    await flush();
    // play a match to the end (first start: matches>=1 -> ad; close it at once)
    lastCall('showStart')![1]('easy');
    await flush();
    f.ad().onClose(false);
    await flush();
    const d = starts().at(-1);
    d.match.result = 'lose';
    d.match.nightTime = 125;
    d.onEnd();
    const [, , onAgain, onMenu] = lastCall('showResult')!;
    onAgain();
    onMenu(); // user changes mind before the ad overlay appears
    await flush();
    expect(qa.hud.screen[0]).toBe('showStart');
    const before = starts().length;
    f.ad().onClose(true);
    await flush();
    expect(starts().length).toBe(before + 1); // match started from the menu
    expect(qa.hud.screen).toBeNull(); // menu hidden by start()
  });

  it('double tap "Попробовать!" on the unlock preview (from menu) -> preview committed twice and two starts', async () => {
    const f = makeSdk({ local: snapshot(10, returning({ matches: 1, unlocks: { pumpkin: 'justUnlocked' } })) });
    await bootMain(f.sdk);
    await flush();
    spritesDone();
    await flush();
    const [kind, onTry] = lastCall('showUnlock')!;
    expect(kind).toBe('pumpkin');
    const revBefore = JSON.parse(f.safe.get('midnight-door-progress')!).rev;
    onTry();
    onTry();
    await flush();
    expect(JSON.parse(f.safe.get('midnight-door-progress')!).rev).toBe(revBefore + 2);
    f.ad().onClose(true);
    await flush();
    expect(starts().length).toBe(2);
  });

  it('first launch on a new device, cloud answers after the 4 s boot timeout while sprites still load -> menu drawn OVER the running tutorial', async () => {
    let answer!: (v: Any) => void;
    const f = makeSdk({ authorized: true, getData: () => new Promise((r) => (answer = r)) });
    await bootMain(f.sdk);
    await vi.advanceTimersByTimeAsync(4000); // boot gives up on the cloud -> tutorial path, waiting for sprites
    expect(qa.hud.calls.some((c: Any) => c[0] === 'showStart')).toBe(false);
    answer({ save: snapshot(40, returning()) }); // cloud arrives: tutorial 'done', 3 matches
    await flush();
    expect(qa.hud.screen[0]).toBe('showStart'); // onReplaced -> showMenu (scene not active yet)
    spritesDone();
    await flush();
    // ...and then startTutorial() resumes and starts the tutorial scene underneath the menu
    expect(starts().length).toBe(1);
    expect(starts()[0].match.opts.tutorial).toBe(true);
    expect(qa.game.scene.isActive('game')).toBe(true);
    expect(qa.hud.screen[0]).toBe('showStart'); // menu card still on top of a live tutorial
  });

  it('returning player taps a difficulty before sprites load, late cloud arrives -> menu drawn over the running match', async () => {
    let answer!: (v: Any) => void;
    const f = makeSdk({ local: snapshot(10, returning()), authorized: true, getData: () => new Promise((r) => (answer = r)) });
    await bootMain(f.sdk);
    await vi.advanceTimersByTimeAsync(4000);
    lastCall('showStart')![1]('hard');
    await flush();
    f.ad().onClose(false);
    await flush(); // start() now awaits whenLoaded()
    answer({ save: snapshot(50, returning({ matches: 7 })) });
    await flush();
    expect(qa.hud.screen[0]).toBe('showStart');
    spritesDone();
    await flush();
    expect(starts().at(-1).match.opts.difficulty).toBe('hard');
    expect(qa.hud.screen[0]).toBe('showStart'); // menu over a live match
  });

  it('"Сохранить" (cloud sign-in) button stays in the menu after the cloud attached via retry', async () => {
    let authed = false;
    let fail = true;
    const f = makeSdk({
      local: snapshot(10, returning()),
      authorized: () => authed,
      getData: async () => {
        if (fail) throw new Error('network');
        return { save: snapshot(99, returning({ matches: 9 })) };
      },
    });
    await bootMain(f.sdk);
    await flush();
    const meta = lastCall('showStart')![3];
    expect(typeof meta.onCloud).toBe('function');
    authed = true;
    meta.onCloud();
    await flush();
    expect(lastCall('toastScreen')![0]).toBe('Не получилось, попробуй позже');
    fail = false;
    await vi.advanceTimersByTimeAsync(30_000); // attach retry succeeds, cloud replaces progress
    expect(f.player.getData).toHaveBeenCalledTimes(2);
    const metaAfter = lastCall('showStart')![3];
    expect(metaAfter.coins).toBe(100);
    expect(typeof metaAfter.onCloud).toBe('function'); // button still offered although already synced
  });

  it('cloud refresh on tab return while the shop is open kicks the player to the menu', async () => {
    const f = makeSdk({ local: snapshot(10, returning()), authorized: true, getData: async () => ({ save: snapshot(10, returning()) }) });
    await bootMain(f.sdk);
    await flush();
    lastCall('showStart')![3].onShop();
    expect(qa.hud.screen[0]).toBe('showShop');
    await vi.advanceTimersByTimeAsync(61_000);
    f.player.getData.mockImplementation(async () => ({ save: snapshot(20, returning({ matches: 4 })) }));
    for (const fn of docListeners['visibilitychange'] ?? []) fn(); // visible again
    await flush();
    expect(qa.hud.screen[0]).toBe('showStart');
  });

  it('cloud replaced while on the result screen: preview "Попробовать!" marks the DETACHED unlocks object -> preview reappears', async () => {
    const f = makeSdk({ local: snapshot(10, returning({ matches: 0 })), authorized: true, getData: async () => ({ save: snapshot(10, returning({ matches: 0 })) }) });
    await bootMain(f.sdk);
    await flush();
    spritesDone();
    await flush();
    lastCall('showStart')![1]('easy'); // matches 0 -> no ad
    await flush();
    const d = starts().at(-1);
    d.match.result = 'lose';
    d.match.nightTime = 60;
    d.onEnd(); // pumpkin -> justUnlocked
    const [, fresh, , onMenu] = lastCall('showResult')!;
    expect(fresh).toBe('pumpkin');
    onMenu(); // -> preview screen (scene still active)
    const [, onTry, previewMenu] = lastCall('showUnlock')!;
    // other device pushed newer save with the same pending preview; tab regains focus > 60 s later
    await vi.advanceTimersByTimeAsync(61_000);
    f.player.getData.mockImplementation(async () => ({ save: snapshot(500, returning({ matches: 1, unlocks: { pumpkin: 'justUnlocked' } })) }));
    for (const fn of docListeners['visibilitychange'] ?? []) fn();
    await flush();
    expect(qa.hud.screen[0]).toBe('showUnlock'); // scene active -> onReplaced did not redraw
    previewMenu(); // "В меню": marks seen on the stale object, then showMenu()
    await flush();
    expect(qa.hud.screen[0]).toBe('showUnlock'); // the same preview again
    void onTry;
  });
});
